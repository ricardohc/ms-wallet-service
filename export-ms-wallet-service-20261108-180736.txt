    public Mono<BigDecimal> getPrimaryAccountBalance(

package com.banking.wallet.client.impl;

import com.banking.wallet.client.AccountValidationClient;
import com.banking.wallet.config.ExternalServicesProperties;
import com.banking.wallet.exception.WalletConflictException;
import com.banking.wallet.exception.WalletUnprocessableException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.web.reactive.function.client.WebClient;
import org.springframework.web.reactive.function.client.WebClientResponseException;
import reactor.core.publisher.Mono;

/**
 * Implementacion del cliente de validacion de cuentas contra ms-account-service.
 * Verifica que el accountType no sea FIXED_TERM.
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class AccountValidationClientImpl implements AccountValidationClient {

    private final WebClient.Builder webClientBuilder;
    private final ObjectMapper objectMapper;
    private final ExternalServicesProperties properties;

    @Override
    public Mono<Void> validateLinkableAccount(UUID accountId) {
        String url = properties.getAccountBaseUrl() + "/api/v1/accounts/" + accountId;
        log.debug("Validating linked account {} against ms-account-service", accountId);
        return webClientBuilder.build()
                .get()
                .uri(url)
                .header("Accept", "application/json")
                // Headers obligatorios exigidos por ms-account-service (OpenAPI)
                .header("X-Correlation-ID", UUID.randomUUID().toString())
                .header("X-Channel-ID", "MS-WALLET-SERVICE")
                .retrieve()
                .bodyToMono(String.class)
                .flatMap(body -> {
                    try {
                        JsonNode node = objectMapper.readTree(body);
                        String accountType = node.path("accountType").asText(null);
                        if ("FIXED_TERM".equals(accountType)) {
                            return Mono.error(new WalletConflictException(
                                    "Fixed-term accounts are not compatible with wallet linking"));
                        }
                        return Mono.empty();
                    } catch (Exception ex) {
                        log.error("Failed to parse account response: {}", ex.getMessage());
                        return Mono.error(new WalletUnprocessableException(
                                "Unable to validate linked account"));
                    }
                })
                .onErrorResume(WebClientResponseException.NotFound.class,
                        ex -> Mono.error(new WalletUnprocessableException(
                                "Linked account not found: " + accountId)))
                .onErrorResume(ex -> {
                    if (ex instanceof WalletConflictException
                            || ex instanceof WalletUnprocessableException) {
                        return Mono.error(ex);
                    }
                    log.error("Failed to validate account {}: {}", accountId, ex.getMessage());
                    return Mono.error(new WalletUnprocessableException(
                            "Unable to validate linked account: " + accountId));
                })
                .then();
    }
}

package com.banking.wallet.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Profile;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.web.server.SecurityWebFilterChain;

@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    /**
     * Perfil dev: permite todo, sin JWT. Útil para Postman/curl local
     * mientras no exista un Identity Provider real (Keycloak/Okta).
     * NUNCA usar en producción.
     */
    @Bean
    @Profile("dev")
    public SecurityWebFilterChain devSecurityWebFilterChain(ServerHttpSecurity http) {
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().permitAll())
                .build();
    }

    /**
     * Perfil no-dev: OAuth2 Resource Server con JWT real.
     * El `phone_number` se toma del token emitido por el IdP.
     */
    @Bean
    @Profile("!dev")
    public SecurityWebFilterChain jwtSecurityWebFilterChain(ServerHttpSecurity http) {
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2.jwt(jwt -> { }))
                .build();
    }
}

package com.banking.wallet.controller;

import com.banking.wallet.api.WalletApi;
import com.banking.wallet.dto.BalanceResponse;
import com.banking.wallet.dto.OperationAcceptedResponse;
import com.banking.wallet.dto.WalletCreateRequest;
import com.banking.wallet.dto.WalletResponse;
import com.banking.wallet.dto.WalletTransferRequest;
import com.banking.wallet.dto.WalletUpdateRequest;
import com.banking.wallet.service.WalletService;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

/**
 * Implementacion del contrato OpenAPI (WalletApi). Reactivo con Reactor,
 * sin block().
 *
 * NOTA: Se elimino la validacion de identidad via JWT en los endpoints
 * createWalletTransfer y findWalletBalance. Ahora el origen de la
 * transferencia viaja en el body (sourcePhoneNumber) y el phoneNumber
 * del balance viaja como query param. Esto es solo para pruebas locales.
 */
@RestController
@RequiredArgsConstructor
@Slf4j
public class WalletApiImpl implements WalletApi {

    private final WalletService walletService;

    @Override
    public Mono<ResponseEntity<WalletResponse>> createWallet(
            String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<WalletCreateRequest> walletCreateRequest, ServerWebExchange exchange) {
        return walletCreateRequest
                .flatMap(req -> walletService.create(req, xTransactionID))
                .map(r -> ResponseEntity.status(HttpStatus.CREATED).body(r));
    }

    @Override
    public Mono<ResponseEntity<WalletResponse>> findWalletById(
            UUID walletId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return walletService.findById(walletId)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<WalletResponse>> updateWallet(
            UUID walletId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, Mono<WalletUpdateRequest> walletUpdateRequest,
            ServerWebExchange exchange) {
        return walletUpdateRequest
                .flatMap(req -> walletService.update(walletId, req, xTransactionID))
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<Void>> deactivateWallet(
            UUID walletId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, ServerWebExchange exchange) {
        return walletService.deactivate(walletId, xTransactionID)
                .thenReturn(ResponseEntity.noContent().build());
    }

    @Override
    public Mono<ResponseEntity<OperationAcceptedResponse>> createWalletTransfer(
            String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<WalletTransferRequest> walletTransferRequest, ServerWebExchange exchange) {
        return walletTransferRequest
                .flatMap(req -> walletService.transfer(
                        req, req.getSourcePhoneNumber(), xTransactionID))
                .map(operationId -> {
                    OperationAcceptedResponse r = new OperationAcceptedResponse();
                    r.setOperationId(operationId);
                    r.setStatus("ACCEPTED");
                    return ResponseEntity.status(HttpStatus.ACCEPTED).body(r);
                });
    }

    @Override
    public Mono<ResponseEntity<BalanceResponse>> findWalletBalance(
            String phoneNumber, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return walletService.findBalance(phoneNumber)
                .map(ResponseEntity::ok);
    }
}

package com.banking.wallet.event;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.core.reactive.ReactiveKafkaProducerTemplate;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

/**
 * Productor de eventos de dominio hacia banking.wallet.events.v1.
 * Los eventos de movimiento se formalizan en el catalogo antes de publicarse.
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class WalletEventPublisher {

    private static final String TOPIC = "banking.wallet.events.v1";

    private final ReactiveKafkaProducerTemplate<String, WalletEvent> kafkaTemplate;

    public Mono<Void> publish(WalletEvent event) {
        return kafkaTemplate.send(TOPIC, event.getWalletId().toString(), event)
                .doOnSuccess(ignored -> log.info("Wallet event published: {}", event.getEventType()))
                .then();
    }
}

package com.banking.wallet.exception;

import com.banking.wallet.dto.ErrorResponse;
import java.time.OffsetDateTime;
import java.util.List;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.bind.support.WebExchangeBindException;
import org.springframework.web.server.ServerWebExchange;
import org.springframework.web.server.ServerWebInputException;
import reactor.core.publisher.Mono;

@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    @ExceptionHandler(WebExchangeBindException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleValidation(
            WebExchangeBindException ex, ServerWebExchange exchange) {
        List<String> details = ex.getFieldErrors().stream()
                .map(err -> err.getField() + ": " + err.getDefaultMessage())
                .toList();
        return build(HttpStatus.BAD_REQUEST, "WALLET-400", "Invalid request",
                exchange, details);
    }

    @ExceptionHandler(ServerWebInputException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleInput(
            ServerWebInputException ex, ServerWebExchange exchange) {
        Throwable cause = ex.getCause();
        String detail = cause != null ? cause.getMessage() : ex.getReason();
        if (detail == null) {
            detail = "Invalid request parameter or body";
        }
        if (detail.length() > 300) {
            detail = detail.substring(0, 300) + "...";
        }
        log.warn("Input error: {}", detail);
        return build(HttpStatus.BAD_REQUEST, "WALLET-400", detail,
                exchange, List.of(detail));
    }

    @ExceptionHandler(BadRequestException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleBadRequest(
            BadRequestException ex, ServerWebExchange exchange) {
        return build(HttpStatus.BAD_REQUEST, "WALLET-400", ex.getMessage(),
                exchange, null);
    }

    @ExceptionHandler(WalletNotFoundException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleNotFound(
            WalletNotFoundException ex, ServerWebExchange exchange) {
        return build(HttpStatus.NOT_FOUND, "WALLET-404", ex.getMessage(), exchange, null);
    }

    @ExceptionHandler(WalletConflictException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleConflict(
            WalletConflictException ex, ServerWebExchange exchange) {
        return build(HttpStatus.CONFLICT, "WALLET-409", ex.getMessage(), exchange, null);
    }

    @ExceptionHandler(WalletForbiddenException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleForbidden(
            WalletForbiddenException ex, ServerWebExchange exchange) {
        return build(HttpStatus.FORBIDDEN, "WALLET-403", ex.getMessage(), exchange, null);
    }

    @ExceptionHandler(WalletUnprocessableException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleUnprocessable(
            WalletUnprocessableException ex, ServerWebExchange exchange) {
        return build(HttpStatus.UNPROCESSABLE_ENTITY, "WALLET-422",
                ex.getMessage(), exchange, null);
    }

    @ExceptionHandler(Exception.class)
    public Mono<ResponseEntity<ErrorResponse>> handleGeneric(
            Exception ex, ServerWebExchange exchange) {
        log.error("Unexpected error", ex);
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "WALLET-500",
                "Internal error", exchange, null);
    }

    private Mono<ResponseEntity<ErrorResponse>> build(HttpStatus status, String code,
                                                      String message, ServerWebExchange exchange, List<String> details) {
        ErrorResponse body = new ErrorResponse();
        body.setCode(code);
        body.setMessage(message);
        body.setTimestamp(OffsetDateTime.now());
        body.setCorrelationId(exchange.getRequest().getHeaders()
                .getFirst("X-Correlation-ID"));
        body.setDetails(details);
        return Mono.just(ResponseEntity.status(status).body(body));
    }
}

package com.banking.wallet.service.impl;

import com.banking.wallet.client.AccountBalanceClient;
import com.banking.wallet.client.AccountValidationClient;
import com.banking.wallet.client.TransactionServiceClient;
import com.banking.wallet.document.WalletDocument;
import com.banking.wallet.dto.WalletCreateRequest;
import com.banking.wallet.dto.WalletResponse;
import com.banking.wallet.dto.WalletStatus;
import com.banking.wallet.dto.WalletTransferRequest;
import com.banking.wallet.dto.WalletUpdateRequest;
import com.banking.wallet.exception.WalletConflictException;
import com.banking.wallet.exception.WalletForbiddenException;
import com.banking.wallet.exception.WalletNotFoundException;
import com.banking.wallet.exception.WalletUnprocessableException;

import com.banking.wallet.event.WalletEventPublisher;
import com.banking.wallet.mapper.WalletMapper;
import com.banking.wallet.repository.WalletRepository;
import com.banking.wallet.service.WalletService;
import com.banking.wallet.service.WalletTransferCommand;
import java.math.BigDecimal;
import java.time.Instant;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;
import reactor.core.publisher.Mono;

/**
 * Logica de negocio del monedero Yanki.
 * Reglas: unicidad documento/telefono, saldo propio sin tarjeta (fuente de
 * verdad), con tarjeta balance local = 0.00 y fuente de verdad ms-account.
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class WalletServiceImpl implements WalletService {

    private static final String SUPPORTED_CURRENCY = "PEN";

    private final WalletRepository repository;
    private final WalletMapper mapper;
    private final WalletEventPublisher eventPublisher;
    private final TransactionServiceClient transactionServiceClient;
    private final AccountBalanceClient accountBalanceClient;
    private final AccountValidationClient accountValidationClient;

    @Override
    public Mono<WalletResponse> create(WalletCreateRequest request, UUID transactionId) {
        return repository.existsByDocumentTypeAndDocumentNumber(
                        request.getDocumentType().getValue(), request.getDocumentNumber())
                .flatMap(exists -> exists
                        ? Mono.error(new WalletConflictException(
                        "Wallet already exists for document"))
                        : repository.existsByPhoneNumber(request.getPhoneNumber()))
                .flatMap(phoneExists -> phoneExists
                        ? Mono.error(new WalletConflictException(
                        "Wallet already exists for phone number"))
                        : persistNewWallet(request, transactionId))
                .onErrorResume(DuplicateKeyException.class,
                        e -> Mono.error(new WalletConflictException(
                                "Wallet already exists for document or phone")));
    }

    private Mono<WalletResponse> persistNewWallet(WalletCreateRequest request, UUID transactionId) {
        WalletDocument doc = mapper.newWallet(
                request.getDocumentType().getValue(),
                request.getDocumentNumber(),
                request.getPhoneNumber(),
                request.getImei(),
                request.getEmail());
        return repository.save(doc)
                .flatMap(saved -> publishEvent("WalletCreated", saved)
                        .thenReturn(mapper.toResponse(saved)))
                .doOnSuccess(r -> log.info("Wallet created txId={}", transactionId));
    }

    @Override
    public Mono<WalletResponse> findById(UUID walletId) {
        return repository.findByWalletUuid(walletId)
                .switchIfEmpty(Mono.error(new WalletNotFoundException("Wallet not found")))
                .map(mapper::toResponse);
    }

    @Override
    public Mono<WalletResponse> update(UUID walletId, WalletUpdateRequest request,
                                       UUID transactionId) {
        return repository.findByWalletUuid(walletId)
                .switchIfEmpty(Mono.error(new WalletNotFoundException("Wallet not found")))
                .flatMap(doc -> applyUpdate(doc, request))
                .flatMap(repository::save)
                .flatMap(saved -> publishEvent("WalletUpdated", saved)
                        .thenReturn(mapper.toResponse(saved)));
    }

    private Mono<WalletDocument> applyUpdate(WalletDocument doc, WalletUpdateRequest request) {
        if (request.getLinkedDebitCardId() != null) {
            return accountValidationClient.validateLinkableAccount(request.getLinkedDebitCardId())
                    .thenReturn(withLinkedCard(doc, request.getLinkedDebitCardId()));
        }
        if (request.getStatus() != null) {
            doc.setStatus(request.getStatus().getValue());
        }
        return Mono.just(doc);
    }

    private WalletDocument withLinkedCard(WalletDocument doc, UUID cardId) {
        doc.setLinkedDebitCardId(cardId);
        doc.setBalance(new BigDecimal("0.00"));
        return doc;
    }

    @Override
    public Mono<Void> deactivate(UUID walletId, UUID transactionId) {
        return repository.findByWalletUuid(walletId)
                .switchIfEmpty(Mono.error(new WalletNotFoundException("Wallet not found")))
                .flatMap(doc -> {
                    doc.setStatus(WalletStatus.INACTIVE.getValue());
                    return repository.save(doc)
                            .then(publishEvent("WalletDeactivated", doc));
                });
    }

    @Override
    public Mono<UUID> transfer(WalletTransferRequest request, String sourcePhoneNumber,
                               UUID transactionId) {
        return Mono.defer(() -> {
            // --- Validacion 1: moneda soportada ---
            if (!SUPPORTED_CURRENCY.equalsIgnoreCase(request.getCurrency())) {
                return Mono.error(new WalletUnprocessableException(
                        "Unsupported currency. Only PEN is allowed"));
            }

            // --- Validacion 2: source != target ---
            if (sourcePhoneNumber.equals(request.getTargetPhoneNumber())) {
                return Mono.error(new WalletUnprocessableException(
                        "Source phone number and target phone number cannot be the same"));
            }

            BigDecimal amount = new BigDecimal(request.getAmount());

            // --- Buscamos el wallet origen ---
            return repository.findByPhoneNumber(sourcePhoneNumber)
                    .switchIfEmpty(Mono.error(new WalletNotFoundException(
                            "Source wallet not found for phone number")))
                    .flatMap(source -> {
                        if (!WalletStatus.ACTIVE.getValue().equals(source.getStatus())) {
                            return Mono.error(new WalletForbiddenException(
                                    "Wallet is not active"));
                        }

                        // --- Validacion 3: fondos suficientes ---
                        Mono<BigDecimal> balanceMono;
                        if (source.getLinkedDebitCardId() != null) {
                            balanceMono = accountBalanceClient
                                    .getPrimaryAccountBalance(source.getWalletUuid(),
                                            source.getLinkedDebitCardId().toString(), null);
                        } else {
                            balanceMono = Mono.just(source.getBalance() == null
                                    ? BigDecimal.ZERO : source.getBalance());
                        }

                        return balanceMono.flatMap(balance -> {
                            if (balance.compareTo(amount) < 0) {
                                return Mono.error(new WalletUnprocessableException(
                                        "Insufficient funds in source wallet"));
                            }
                            return delegateTransfer(source, request, transactionId);
                        });
                    });
        });
    }

    private Mono<UUID> delegateTransfer(WalletDocument source, WalletTransferRequest request,
                                        UUID transactionId) {
        WalletTransferCommand command = WalletTransferCommand.builder()
                .transactionId(transactionId)
                .sourceWalletId(source.getWalletUuid())
                .targetPhoneNumber(request.getTargetPhoneNumber())
                .amount(new BigDecimal(request.getAmount()))
                .currency(request.getCurrency())
                .build();
        return transactionServiceClient.createTransfer(command);
    }

    @Override
    public Mono<com.banking.wallet.dto.BalanceResponse> findBalance(
            String authenticatedPhoneNumber) {
        return repository.findByPhoneNumber(authenticatedPhoneNumber)
                .switchIfEmpty(Mono.error(new WalletNotFoundException(
                        "Wallet not found for phone number")))
                .flatMap(doc -> {
                    if (doc.getLinkedDebitCardId() != null) {
                        return accountBalanceClient
                                .getPrimaryAccountBalance(doc.getWalletUuid(),
                                        doc.getLinkedDebitCardId().toString(), null)
                                .map(balance -> toBalance(balance, doc.getCurrency()));
                    }
                    return Mono.just(toBalance(doc.getBalance(), doc.getCurrency()));
                });
    }

    private com.banking.wallet.dto.BalanceResponse toBalance(BigDecimal balance, String currency) {
        com.banking.wallet.dto.BalanceResponse r = new com.banking.wallet.dto.BalanceResponse();
        r.setBalance(balance.toPlainString());
        r.setCurrency(currency);
        return r;
    }

    private Mono<Void> publishEvent(String eventType, WalletDocument doc) {
        WalletEvent event = WalletEvent.builder()
                .eventType(eventType)
                .walletId(doc.getWalletUuid())
                .phoneNumber(doc.getPhoneNumber())
                .occurredAt(Instant.now())
                .build();
        return eventPublisher.publish(event).onErrorResume(e -> {
            log.warn("Failed to publish wallet event {}: {}", eventType, e.getMessage());
            return Mono.empty();
        });
    }
}
